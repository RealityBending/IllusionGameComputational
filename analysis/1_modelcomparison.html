# A fit does not know which illusion it was fitted to: its data only has the
# Z-scored predictors, and combine_model.R drops $file. The file name is the one
# place it is recorded (<model>_<illusion>[_suffix].rds), so it is attached to
# the fit here, for model_illusion(). Only needed to compute predictions
# locally (see load_predictions() below).
load_fit <- function(file) {
  illusion <- regmatches(file, regexpr("MullerLyer|Ebbinghaus|VerticalHorizontal", file))
  if (length(illusion) == 0) stop("No illusion in the file name: ", file)
  m <- readRDS(paste0(path, file))
  attr(m, "illusion") <- illusion
  m
}

# The results of server/predictions.R for one model, from
# models/predictions/<name>.rds as written by `./hpc predict` and fetched by
# `./hpc pull predictions`. A model with no such file but a fit in models/ has
# them computed here instead, once, and saved to the same place -- slow, and it
# loads the fit, but it means a fit pulled without its predictions still works.
# Delete a prediction file to have it recomputed.
load_predictions <- function(name) {
  file <- paste0(pred_path, name, ".rds")
  fit_file <- paste0(path, name, ".rds")
  if (!file.exists(file)) {
    if (!file.exists(fit_file)) {
      stop("No predictions and no fit for ", name, ": run `./hpc predict ",
           sub("_[^_]+$", "", name), "` and `./hpc pull predictions`.")
    }
    message("No predictions for ", name, ": computing them from the local fit (slow).")
    fit <- load_fit(basename(fit_file))
    rez <- run_predictions(fit, name, fit_file = fit_file)
    rm(fit)
    gc()
    dir.create(pred_path, showWarnings = FALSE)
    saveRDS(rez, file)
  }
  rez <- readRDS(file)

  if (!identical(rez$meta$version, igc_predictions_version)) {
    stop(name, ": the predictions are format ", rez$meta$version, " and this script reads ",
         igc_predictions_version, ". Re-run `./hpc predict`, or delete ", file,
         " to recompute locally.")
  }
  # A fit re-combined or re-fitted since. The size is compared rather than the
  # time, which a copy does not always keep.
  if (file.exists(fit_file) && !identical(file.size(fit_file), rez$meta$fit_size)) {
    warning(name, ": the predictions were made from another fit than models/", basename(fit_file),
            ". Re-run `./hpc predict`, or delete ", file, " to recompute locally.")
  }
  if (!identical(plot_ranges[[rez$meta$illusion]], rez$meta$plot_range)) {
    warning(name, ": plot_ranges has changed since the predictions were made.")
  }
  rez
}

# Sampling efficiency: bulk ESS per hour of sampling, chain by chain (see
# compute_efficiency() in server/predictions.R for what is counted)
tidy_efficiency <- function(preds) {
  purrr::imap_dfr(preds, \(p, name) mutate(p$efficiency, Model = name)) |>
    mutate(Model = factor(Model, levels = names(preds)))
}

plot_efficiency <- function(preds) {
  efficiency <- tidy_efficiency(preds)

  # Model-level summary: the chains of a model average to one point
  efficiency_means <- efficiency |>
    summarise(.by = Model,
              ESS_per_hour = mean(ESS_per_hour),
              Time_total = mean(Time_total))

  # What each fit was: two models are only comparable here if these match
  caption <- efficiency |>
    summarise(.by = Model,
              label = sprintf("%s: %d chains, %g threads/chain, n = %s",
                              first(Model), n(), first(Threads),
                              format(first(N), big.mark = ","))) |>
    pull(label) |>
    paste(collapse = "\n")

  efficiency |>
    ggplot(aes(x = ESS_per_hour, y = Time_total / 3600, colour = Model)) +
    # One point per chain, and the model's mean on top of them
    geom_point(size = 2.5, alpha = 0.35) +
    geom_point(data = efficiency_means, size = 5, shape = 18) +
    # Both axes log: families differ in cost per gradient by orders of magnitude
    # (the DDM's numerical quadrature is 55x, see cogmod_ddm_cost_issue.md), and
    # on a linear axis the cheap ones collapse onto the origin
    scale_x_log10() +
    scale_y_log10() +
    scale_colour_manual(values = model_colors) +
    labs(
      x = "Effective sample size per hour of sampling (bulk ESS / h, log)",
      y = "Wall clock per chain, warmup included (hours, log)",
      title = "Computational cost",
      subtitle = "One point per chain, diamond is the model's mean. Bottom right is cheap and efficient.",
      caption = caption
    ) +
    theme_minimal() +
    theme(plot.caption = element_text(hjust = 0, colour = "grey40"))
}

# Total compute, three ways, for a rough headline number:
# - Wall clock: the shards of a model run at the same time on the cluster, so
#   what one waited for is the longest chain (queueing, compilation and the
#   combine/loo step are not included).
# - Chains one after the other: the sum of the chain times, i.e. the same run
#   with no parallel chains but still multithreaded within each chain.
# - Single core: that sum times the threads per chain. This assumes within-
#   chain threading scales linearly, which is close to what was measured
#   (4 -> 8 threads halved the time per chain, server/AGENT.md 4.4.1), so read
#   it as an order of magnitude, not a measurement.
table_compute_time <- function(preds) {
  tidy_efficiency(preds) |>
    summarise(.by = Model,
              Chains = n(),
              Threads = first(Threads),
              Cores = n() * first(Threads),
              Wall = max(Time_total) / 3600,
              Serial = sum(Time_total) / 3600,
              SingleCore = sum(Time_total * Threads) / 3600) |>
    mutate(SingleCore_days = SingleCore / 24,
           SingleCore_years = SingleCore / (24 * 365.25)) |>
    gt::gt(rowname_col = "Model") |>
    gt::cols_label(
      Chains = "Chains",
      Threads = "Threads / chain",
      Cores = "Cores used",
      Wall = "Wall clock (h)",
      Serial = "Chains in sequence (h)",
      SingleCore = "Single core (h)",
      SingleCore_days = "Single core (days)",
      SingleCore_years = "Single core (years)"
    ) |>
    gt::fmt_number(columns = c(Wall, Serial, SingleCore, SingleCore_days), decimals = 0) |>
    gt::fmt_number(columns = SingleCore_years, decimals = 2) |>
    gt::tab_spanner(label = "Without parallelisation (estimated)",
                    columns = c(Serial, SingleCore, SingleCore_days, SingleCore_years)) |>
    gt::tab_source_note("Warmup + sampling only. Single core assumes linear scaling of within-chain threading.")
}

# loo was added at combine time and lives in the fit's $criteria$loo, which the
# prediction files carry over. The list names become the model labels.
# loo_compare() needs two models at least, which an illusion only just started
# on does not have yet.
show_loo <- function(preds) {
  if (length(preds) < 2) {
    cat("Only one model so far (", names(preds), "): nothing to compare.\n", sep = "")
    return(invisible(NULL))
  }
  rez_loo <- loo_compare(lapply(preds, \(p) p$loo))
  print(parameters(rez_loo))
  print(report(rez_loo))
}

# What the fit costs against how well it does: the elpd against the time a chain
# took, the y axis of the computational cost plot. The elpd is relative to the
# best model (loo_compare), with the SE of that difference: each model's own SE
# (~800 here) is mostly shared between models fitted to the same trials, and
# would swamp differences that are in fact well resolved.
plot_loo_cost <- function(preds) {
  if (length(preds) < 2) {
    cat("Only one model so far (", names(preds), "): nothing to compare.\n", sep = "")
    return(invisible(NULL))
  }
  # loo 2.10 returns a data frame with the names in a `model` column, earlier
  # versions a matrix with them as row names
  rez_loo <- as.data.frame(loo_compare(lapply(preds, \(p) p$loo)))
  elpd <- data.frame(
    Model = factor(if (is.null(rez_loo$model)) rownames(rez_loo) else rez_loo$model,
                   levels = names(preds)),
    elpd_diff = rez_loo$elpd_diff,
    se_diff = rez_loo$se_diff
  )

  # One point per chain, as in the cost plot: a model has one elpd, so its
  # chains spread along x only
  efficiency <- tidy_efficiency(preds) |>
    mutate(Time_total = Time_total / 3600) |>
    left_join(elpd, by = "Model")
  efficiency_means <- efficiency |>
    summarise(.by = c(Model, elpd_diff, se_diff),
              Time_total = mean(Time_total))

  ggplot(efficiency, aes(x = Time_total, y = elpd_diff, colour = Model)) +
    geom_hline(yintercept = 0, colour = "grey60", linetype = "dashed") +
    geom_point(size = 2.5, alpha = 0.35) +
    geom_linerange(data = efficiency_means,
                   aes(ymin = elpd_diff - se_diff, ymax = elpd_diff + se_diff)) +
    geom_point(data = efficiency_means, size = 5, shape = 18) +
    # Labelled in place: with the colours already shared with every other
    # plot, a legend would only be a lookup. The chains come along with an
    # empty label, which ggrepel then keeps the labels off.
    ggrepel::geom_text_repel(data = bind_rows(mutate(efficiency, Label = ""),
                                              mutate(efficiency_means, Label = as.character(Model))),
                             aes(label = Label),
                             size = 3.5, fontface = "bold", seed = 3, box.padding = 0.5,
                             max.overlaps = Inf) +
    # Log, as in the cost plot
    scale_x_log10() +
    # Headroom above the best model, whose label otherwise has nowhere to go
    # but onto the chains of its neighbours on the zero line
    scale_y_continuous(expand = expansion(mult = c(0.05, 0.15))) +
    scale_colour_manual(values = model_colors, guide = "none") +
    labs(
      x = "Wall clock per chain, warmup included (hours, log)",
      y = "ELPD difference from the best model",
      title = "Performance against computational cost",
      subtitle = paste("Higher is better (dashed: the best model), left is cheaper. Diamond is the model's",
                       "mean over its chains, faint points the chains, bars ±1 SE of the difference.",
                       sep = "\n")
    ) +
    theme_minimal()
}

# All models are checked against the same trials: the observed data is taken
# from the first model's file, so the models must share an illusion. What is
# predicted, and how, is compute_ppc() in server/predictions.R.
plot_ppc <- function(preds, rt_range = c(0, 2.5), binwidth = 0.02, n_points = 512) {

  if (is.null(names(preds))) names(preds) <- paste("Model", seq_along(preds))
  if (length(unique(vapply(preds, \(p) p$meta$illusion, character(1)))) > 1) {
    stop("plot_ppc() compares models fitted to the same illusion.")
  }

  # Observed: histogram of every trial, put on the same defective-density
  # scale. Stored as the count of each distinct (RT, Error), so this is exact.
  observed <- preds[[1]]$ppc$observed
  obs_data <- observed |>
    mutate(bin = floor(RT / binwidth)) |>
    count(Error, bin, wt = n, name = "n") |>
    mutate(
      x = (bin + 0.5) * binwidth,
      y = n / (sum(observed$n) * binwidth) * if_else(Error == 0, 1, -1),
      Response = factor(Error, levels = c(0, 1))
    ) |>
    filter(x <= max(rt_range))

  # Predicted: defective densities, each response's curve scaled by its own
  # rate, so the two areas sum to 1 and correct/error are directly comparable
  # across models. The predicted RTs come binned (1 ms) with the bandwidth
  # density() picked on the raw values, so this is the curve the raw RTs give,
  # to within a bin: density() itself bins its input far more coarsely.
  defective_density <- function(pp) {
    total <- sum(pp$n)
    purrr::map_dfr(c(0, 1), function(r) {
      r_chr <- as.character(r)
      if (pp$n[[r_chr]] < 2) return(NULL)
      d <- pp$counts[pp$counts$Response == r, ]
      dens <- density((d$Bin + 0.5) * pp$bin, weights = d$n / sum(d$n), bw = pp$bw[[r_chr]],
                      from = rt_range[1], to = rt_range[2], n = n_points)
      data.frame(
        Response = factor(r, levels = c(0, 1)),
        x = dens$x,
        y = dens$y * (pp$n[[r_chr]] / total) * if (r == 0) 1 else -1
      )
    })
  }
  pred_data <- purrr::imap_dfr(preds, \(p, nm) mutate(defective_density(p$ppc$predicted), Model = nm))
  pred_data$Model <- factor(pred_data$Model, levels = names(preds))

  ggplot(mapping = aes(x = x, y = y)) +
    geom_col(
      data = obs_data,
      aes(fill = Response),
      width = binwidth, alpha = 0.4
    ) +
    geom_line(
      data = pred_data,
      aes(color = Model, group = interaction(Model, Response)),
      linewidth = 0.9, alpha = 0.9
    ) +
    geom_hline(yintercept = 0, color = "grey30", linewidth = 0.3) +
    scale_fill_manual(
      values = c("0" = "#43A047", "1" = "#F44336"),
      labels = c("0" = "Correct", "1" = "Error")
    ) +
    scale_color_manual(values = model_colors) +
    scale_y_continuous(labels = abs) +
    coord_cartesian(xlim = rt_range) +
    labs(x = "RT", y = "Density", fill = "Observed", color = "Predicted") +
    guides(fill = guide_legend(override.aes = list(alpha = 0.6))) +
    theme_minimal() +
    theme(legend.position = "bottom")
}

# Display names for the parameters, in the order they are plotted. Names are
# the family's own ("mu", "ndt", ...), values are what goes in the panel
# titles. Several parameters mean different things from one family to the next
# - "mu" is a drift rate in the DDM and the log-mean of the correct response in
# the LNR, and "sigmabias" is the variability of the starting point in the DDM
# but its maximum (A) in the racing models - so the map is built per family
# (the brms family name, e.g. "cogmod_ddm", as stored in the prediction files).
param_labels <- function(family) {
  lvls <- c("Error Rate" = "Error Rate",
            "Mean RT" = "Mean RT",
            "mu" = "mu",
            "nuone" = "Nu (Errors)",
            "driftone" = "Drift (Errors)",
            "sigmadrift" = "Drift Variability",
            "sigmazero" = "Sigma (Correct)",
            "sigmaone" = "Sigma (Errors)",
            "boundary" = "Boundary",
            "bias" = "Bias",
            "sigmabias" = "Bias Variability",
            "ndt" = "NDT",
            "sigmandt" = "NDT Variability")
  switch(
    family,
    "cogmod_ddm" = {
      lvls["mu"] <- "Drift"
    },
    # sigmabias is only estimated in LNR-6, where it is the start-point range
    # of the race, as in the other racing families. Each accumulator's
    # parameters are kept together - the correct response's nu and its
    # variability, then the error's - rather than the two nus first.
    "cogmod_lnr" = {
      lvls[c("mu", "sigmazero", "sigmaone", "sigmabias")] <-
        c("Nu (Correct)", "Nu Variability (Correct)", "Nu Variability (Errors)",
          "Start Point Range")
      first <- c("Error Rate", "Mean RT", "mu", "sigmazero", "nuone", "sigmaone")
      lvls <- lvls[c(first, setdiff(names(lvls), first))]
    },
    # Both racing families: mu is the drift of the accumulator for correct
    # responses and driftone that of the other one, and sigmabias is the
    # start-point range A rather than a variability around a bias
    "cogmod_rdm" = {
      lvls[c("mu", "sigmabias")] <- c("Drift (Correct)", "Start Point Range")
    },
    "cogmod_lba2" = {
      lvls[c("mu", "sigmabias", "sigmazero", "sigmaone")] <-
        c("Drift (Correct)", "Start Point Range",
          "Drift SD (Correct)", "Drift SD (Errors)")
    }
  )
  lvls
}

# Rename the parameters and put them in canonical order. Parameter_Order keeps
# that order recoverable once the models are stacked together, where the factor
# levels of families that do not share their parameters get merged in whatever
# order the models happen to come in. It is the position in the full map rather
# than among the parameters this model has, so that it is comparable between
# models of a family that do not estimate the same ones.
format_parameters <- function(pred, family) {
  lvls <- param_labels(family)
  found <- unique(as.character(pred$Parameter))
  # A parameter the map does not know about (a family added later) keeps its
  # own name and goes last, rather than silently becoming an NA panel
  unknown <- setdiff(found, names(lvls))
  if (length(unknown) > 0) lvls <- c(lvls, setNames(unknown, unknown))
  pred$Parameter_Order <- match(as.character(pred$Parameter), names(lvls))
  lvls <- lvls[names(lvls) %in% found]

  pred$Parameter <- factor(unname(lvls[as.character(pred$Parameter)]),
                           levels = unname(lvls))
  pred
}

# The predictions of each of an illusion's models, relabelled, back in raw
# units, and stacked. Adding a model to `models` in the loading chunk is all it
# takes for it to get its own column in every plot below.
tidy_predictions <- function(preds, what) {
  purrr::imap_dfr(preds, \(p, name) {
    p[[what]] |>
      format_parameters(p$meta$family) |>
      # Restore the original scale of the illusion strength and difference
      restore_units(illusion = p$meta$illusion) |>
      mutate(Model = name, Family = p$meta$family)
  }) |>
    mutate(Model = factor(Model, levels = names(preds)))
}

# Parameters against illusion strength, one curve per illusion difficulty
# (compute_curves() in server/predictions.R), one data frame per illusion
preds_curves <- lapply(preds, tidy_predictions, what = "curves")

make_header <- function(label) {
  ggplot() +
    annotate(
      "text",
      x = 0, y = 0,
      label = label,
      fontface = "bold",
      size = 5
    ) +
    coord_cartesian(xlim = c(-1, 1), ylim = c(-1, 1)) +
    theme_void()
}

# Height of the column-header row, in units of one panel row. A figure that
# puts something alongside the grid has to skip the headers to line up with the
# panels, so this is a shared constant rather than a default buried in the
# function below.
parameter_grid_header <- 0.35

# Which parameter goes in which row of which column. Models of one family share
# their parameters, so within a family rows are matched by name: the family's
# rows are every parameter any of its models has, in canonical order, and a
# model that does not estimate one (DDM-4's drift variability, say) gets a gap
# there. Families do not share them, so across families rows cannot be matched
# by name: each family's rows are packed from the top (which lines up the drift
# rates and the nus, both "mu"), and the non-decision time - the one parameter
# every family has - is anchored to the bottom row. Shorter columns get NA in
# between, drawn as spacers, so the columns stay aligned however many
# parameters they have.
parameter_grid_rows <- function(data, anchor_bottom = c("NDT", "NDT Variability")) {
  data$Model <- droplevels(factor(data$Model))
  # Data without a family (made by hand, say) treats each model as its own
  if (is.null(data$Family)) data$Family <- as.character(data$Model)

  in_order <- function(d) unique(as.character(d$Parameter[order(d$Parameter_Order)]))
  slots <- lapply(levels(data$Model), function(model) {
    family_rows <- in_order(data[data$Family == data$Family[data$Model == model][1], ])
    own <- in_order(data[data$Model == model, ])
    rows <- ifelse(family_rows %in% own, family_rows, NA_character_)
    list(top = rows[!family_rows %in% anchor_bottom],
         bottom = rows[family_rows %in% anchor_bottom])
  })
  names(slots) <- levels(data$Model)
  n_top <- max(lengths(lapply(slots, `[[`, "top")))
  n_bottom <- max(lengths(lapply(slots, `[[`, "bottom")))

  lapply(slots, function(s) {
    c(s$top, rep(NA_character_, n_top - length(s$top)),
      rep(NA_character_, n_bottom - length(s$bottom)), s$bottom)
  })
}

# One column per model, one row per parameter, with panel(model, parameter)
# drawing each cell. The cells go into a single grid, filled column by column,
# rather than one sub-layout per model put side by side: patchwork only aligns
# panels within a layout, so separate columns let each one's rows drift with
# its own titles and legends.
plot_parameter_grid <- function(data, panel,
                                anchor_bottom = c("NDT", "NDT Variability"),
                                header_height = parameter_grid_header) {
  rows_by_model <- parameter_grid_rows(data, anchor_bottom)
  cells <- unlist(Map(function(model, rows) {
    c(list(make_header(model)),
      lapply(rows, function(p) if (is.na(p)) plot_spacer() else panel(model, p)))
  }, names(rows_by_model), rows_by_model), recursive = FALSE)

  wrap_plots(cells, ncol = length(rows_by_model), byrow = FALSE,
             heights = c(header_height, rep(1, length(rows_by_model[[1]]))))
}

# The curves of one illusion's models, from one element of preds_curves
plot_curves <- function(preds_curves) {
  # Prepare data
  curves <- preds_curves |>
    filter(!Parameter %in% c("Error Rate", "Mean RT")) |>
    # Rounded, or the levels print their floating-point error (0.0500000000001)
    mutate(Illusion_Difference = factor(round(Illusion_Difference, 5)))

  # Colour scale
  curve_cols <- colorRampPalette(
    c("#FF5722", "#FFC107", "#8BC34A", "#4CAF50")
  )(nlevels(curves$Illusion_Difference))

  # Function for individual panels
  make_curve <- function(model, parameter) {
    ggplot(
      filter(curves, Model == model, Parameter == parameter),
      aes(Illusion_Strength, Predicted, colour = Illusion_Difference)
    ) +
      geom_line(linewidth = 1.3) +
      scale_colour_manual(values = curve_cols) +
      labs(title = parameter, color = "Task Difficulty") +
      theme_minimal() +
      theme(
        plot.title = element_text(
          hjust = 0.5,
          size = 11,
          margin = margin(b = 4)
        ),
        axis.title = element_blank()
      )
  }

  plot_parameter_grid(curves, make_curve) +
    plot_layout(guides = "collect") &
    theme(
      legend.position = "right",
      plot.margin = margin(4, 4, 4, 4)
    )
}

# Two ways of reading the values off the map: a colour bar per panel, or, as on
# an elevation map, the value written on a handful of the contour lines (which
# makes the colour bars redundant, so they are dropped)
heatmap_labels <- TRUE

# The region of the space that might be psychometrically optimal, outlined on
# every panel, in raw units. Only worked out for the Muller-Lyer so far: an
# illusion without one gets no outline.
optimal_regions <- list(
  MullerLyer = list(strength = c(-30, 30), difference = c(0.1, 0.4))
)

# Function for individual panels. The illusion is passed along for the
# outline, which is in its raw units.
make_heatmap <- function(preds_heatmaps, model, parameter, illusion) {
  d <- filter(preds_heatmaps, Model == model, Parameter == parameter)
  region <- optimal_regions[[illusion]]
  contours <- if (heatmap_labels) {
    list(
      # Few enough lines that the labels have room: 20 would overlap in panels
      # this short. Breaks strictly inside the range, as pretty() can overshoot.
      geomtextpath::geom_textcontour(
        # The label is given explicitly: the default rounds it, which writes
        # 0.4 on every NDT contour from 0.37 to 0.41. The sign is a true minus
        # (U+2212): geomtextpath 0.2.0 silently drops an ASCII hyphen, which
        # labels -3 as 3 and makes the colour scale look reversed.
        aes(z = Predicted,
            label = after_stat(sub("^-", "−", format(level, trim = TRUE, drop0trailing = TRUE)))),
        breaks = (\(b) b[b > min(d$Predicted) & b < max(d$Predicted)])(pretty(d$Predicted, n = 10)),
        size = 2.5, linecolour = "white", textcolour = "white", linewidth = 0.4, alpha = 0.5,
        padding = grid::unit(0.5, "mm"), straight = TRUE
      ),
      guides(fill = "none")
    )
  } else {
    geom_contour(aes(z = Predicted), color = "white", alpha = 0.3, linewidth = 0.5, bins = 20)
  }

  ggplot(d, aes(x = Illusion_Strength, y = Illusion_Difference)) +
    geom_tile(aes(fill = Predicted)) +
    contours +
    geom_vline(xintercept = 0, color = "white", linewidth = 0.3, linetype = "dashed") +
    (if (!is.null(region)) {
      annotate("rect", xmin = region$strength[1], xmax = region$strength[2],
               ymin = region$difference[1], ymax = region$difference[2],
               fill = NA, color = "red", linewidth = 0.5, linetype = "dotted")
    }) +
    scale_fill_viridis_c(option = "turbo") +
    scale_x_continuous(expand = c(0, 0)) +
    scale_y_continuous(expand = c(0, 0)) +
    labs(title = parameter, fill = NULL) +
    theme_minimal() +
    theme(
      plot.title = element_text(hjust = 0.5, size = 11, margin = margin(b = 4)),
      axis.title = element_blank(),
      legend.key.width = grid::unit(0.25, "cm"),
      # A null height stretches the colour bar to the height of its panel, so
      # it can neither overflow into the panel below nor sit off-centre
      legend.key.height = grid::unit(1, "null"),
      legend.ticks = element_blank(),
      legend.text = element_text(size = 7)
    )
}

# Every panel shares the same axes, so the titles are dropped from the panels
# (see make_heatmap) and written once around the whole grid, next to a strip of
# stimuli showing what each axis means. patchwork has no shared axis title: the
# titles and the images each get their own row/column of the layout (the
# plot.tag route needs two nested wrap_elements, which renders the grid blank in
# patchwork 1.3.2).
axis_title <- function(label, rot = 0) {
  wrap_elements(grid::textGrob(label, rot = rot, gp = grid::gpar(fontsize = 13)))
}

# Stimuli along each axis, rendered by illustrations/make_illustrations.py
# (re-run it if the stimulus bounds change) as <illusion in lower case>_*.png.
# Both series are numbered low value to high, so they run in the direction of
# their axis: x left to right, y bottom to top, hence the reversal below. Only
# the Muller-Lyer has them so far; an illusion without them gets the grid and
# the axis titles alone.
illustrations_dir <- if (dir.exists("illustrations")) "illustrations" else "analysis/illustrations"
illustration_file <- function(illusion, name) {
  file.path(illustrations_dir, paste0(tolower(illusion), "_", name, ".png"))
}

# Every stimulus is drawn at the same size on both axes. The images already
# share their pixel dimensions (make_illustrations.py crops them with one box),
# so an explicit height is all it takes -- with the width left out, rasterGrob
# takes it from the aspect ratio. Letting each raster fill its cell instead
# would size the y-axis ones by the column width and the x-axis ones by the row
# height, which come out different once patchwork has taken its margins.
stim_h <- 1.55  # height of every stimulus, in inches
illustration <- function(illusion, name, vjust = "centre") {
  wrap_elements(grid::rasterGrob(
    png::readPNG(illustration_file(illusion, name)),
    # capped at the cell, so a figure too short for the strip shrinks it rather
    # than clipping it
    height = grid::unit.pmin(grid::unit(stim_h, "in"), grid::unit(1, "npc")),
    y = grid::unit(c(top = 1, centre = 0.5, bottom = 0)[[vjust]], "npc"),
    just = c("centre", vjust),
    interpolate = TRUE
  ))
}

# The heatmaps of one illusion's models, from one element of preds_heatmaps
plot_heatmaps <- function(preds_heatmaps, illusion) {
  # Each parameter is on its own scale, so unlike the curves the fill guides
  # cannot be collected: every panel keeps its own colour bar
  heatmaps <- plot_parameter_grid(
    preds_heatmaps,
    \(model, parameter) make_heatmap(preds_heatmaps, model, parameter, illusion)
  ) &
    theme(plot.margin = margin(4, 4, 4, 4))

  # The y-axis column starts at the first panel, not at the column headers. The
  # row count is recomputed rather than carried out of plot_parameter_grid(): a
  # stale value here silently collapses the heights to an even split, which puts
  # the strip in the bottom half of the grid.
  grid_rows <- length(parameter_grid_rows(preds_heatmaps)[[1]])
  below_header <- function(p) {
    wrap_plots(plot_spacer(), p, ncol = 1, heights = c(parameter_grid_header, grid_rows))
  }

  # The strip and title rows/columns are sized in inches and the grid takes the
  # rest ("null"). Relative sizes would be read against the panels rather than
  # the cells, which leaves the strips adrift from the grid they annotate.
  if (!file.exists(illustration_file(illusion, "strength1"))) {
    return(wrap_plots(
      below_header(axis_title("Task Difficulty", rot = 90)), heatmaps,
      plot_spacer(), axis_title("Illusion Strength"),
      ncol = 2,
      widths = grid::unit(c(0.3, 1), c("in", "null")),
      heights = grid::unit(c(1, 0.3), c("null", "in"))
    ))
  }

  stim_aspect <- local({
    d <- dim(png::readPNG(illustration_file(illusion, "strength1")))
    d[2] / d[1]
  })
  # The outer two mark the ends of the y axis, so they sit flush with the top
  # and the bottom of the panel block rather than floating in the middle of
  # their cell
  y_strip <- wrap_plots(
    Map(\(name, vjust) illustration(illusion, name, vjust),
        paste0("difference", 3:1), c("top", "centre", "bottom")),
    ncol = 1
  )
  x_strip <- wrap_plots(lapply(paste0("strength", 1:3), \(name) illustration(illusion, name)), nrow = 1)

  wrap_plots(
    below_header(axis_title("Task Difficulty", rot = 90)), below_header(y_strip), heatmaps,
    plot_spacer(), plot_spacer(), x_strip,
    plot_spacer(), plot_spacer(), axis_title("Illusion Strength"),
    ncol = 3,
    widths = grid::unit(c(0.3, stim_h * stim_aspect + 0.2, 1), c("in", "in", "null")),
    heights = grid::unit(c(1, stim_h + 0.2, 0.3), c("null", "in", "in"))
  )
}
