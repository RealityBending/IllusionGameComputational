# A fit does not know which illusion it was fitted to: its data only has the
# Z-scored predictors, and combine_model.R drops $file. The file name is the one
# place it is recorded (<model>_<illusion>[_suffix].rds), so it is attached to
# the fit here, for model_illusion(). Only needed to compute predictions
# locally (see load_predictions() below).
load_fit <- function(file, dir = path) {
  illusion <- regmatches(file, regexpr("MullerLyer|Ebbinghaus|VerticalHorizontal", file))
  if (length(illusion) == 0) stop("No illusion in the file name: ", file)
  m <- readRDS(paste0(dir, file))
  attr(m, "illusion") <- illusion
  m
}

# The results of server/predictions.R for one model, from
# models/predictions/<name>.rds as written by `./hpc predict` and fetched by
# `./hpc pull predictions` (models_stanli/ for the stanli fit). A model with no
# such file but a fit in models/ has them computed here instead, once, and
# saved to the same place -- slow, and it loads the fit, but it means a fit
# pulled without its predictions still works. Delete a prediction file to have
# it recomputed. The sampler is recorded in $meta$sampler, which the files
# themselves do not have.
load_predictions <- function(name, sampler = "cmdstanr") {
  dir <- model_dirs[[sampler]]
  hpc <- if (sampler == "stanli") "IGC_BACKEND=stanli ./hpc" else "./hpc"
  file <- paste0(dir, "predictions/", name, ".rds")
  fit_file <- paste0(dir, name, ".rds")
  if (!file.exists(file)) {
    if (!file.exists(fit_file)) {
      stop("No predictions and no fit for ", name, " (", sampler, "): run `", hpc, " predict ",
           sub("_[^_]+$", "", name), "` and `", hpc, " pull predictions`.")
    }
    message("No predictions for ", name, " (", sampler, "): computing them from the local fit (slow).")
    fit <- load_fit(basename(fit_file), dir)
    rez <- run_predictions(fit, name, fit_file = fit_file)
    rm(fit)
    gc()
    dir.create(paste0(dir, "predictions/"), recursive = TRUE, showWarnings = FALSE)
    saveRDS(rez, file)
  }
  rez <- readRDS(file)

  if (!identical(rez$meta$version, igc_predictions_version)) {
    stop(name, " (", sampler, "): the predictions are format ", rez$meta$version,
         " and this script reads ", igc_predictions_version, ". Re-run `", hpc,
         " predict`, or delete ", file, " to recompute locally.")
  }
  # A fit re-combined or re-fitted since. The size is compared rather than the
  # time, which a copy does not always keep.
  if (file.exists(fit_file) && !identical(file.size(fit_file), rez$meta$fit_size)) {
    warning(name, ": the predictions were made from another fit than ", fit_file,
            ". Re-run `", hpc, " predict`, or delete ", file, " to recompute locally.")
  }
  # Every curve and heatmap is drawn over the reliable predictor space
  # (plot_ranges in server/predictions.R), whatever range the file was made
  # over: a wider one is cropped to it. A file that does not cover it all (the
  # Müller-Lyer files made before 2026-11-25 start at a difference of 0.05,
  # not 0.045) is drawn over what it has.
  if (!covers_plot_range(rez$meta$plot_range, rez$meta$illusion)) {
    message(name, " (", sampler, "): the predictions do not cover all of plot_ranges. Re-run `",
            hpc, " predict` for the whole reliable space.")
  }
  rez$curves <- crop_to_plot_range(rez$curves, rez$meta$illusion)
  rez$heatmaps <- crop_to_plot_range(rez$heatmaps, rez$meta$illusion)
  rez$meta$sampler <- sampler
  rez
}

# For the tables: the same colours behind the model names, and a way to fill
# the cells of a column. The text is white on a dark fill and black on a light
# one.
text_on <- function(fill) {
  lum <- apply(grDevices::col2rgb(fill), 2, \(x) sum(c(0.299, 0.587, 0.114) * x)) / 255
  ifelse(lum < 0.55, "white", "black")
}
fill_cells <- function(tab, j, fill) {
  for (i in seq_along(fill)) {
    tab <- tinytable::style_tt(tab, i = i, j = j, background = fill[i], color = text_on(fill[i]))
  }
  tab
}
model_fill <- function(labels) {
  fill <- model_colors[as.character(labels)]
  fill[is.na(fill)] <- "#BDBDBD"
  unname(fill)
}

Convergence

A model is only worth comparing if its chains agree. “Population” is the smooth coefficients, their SDs and the group-level SDs. The participant offsets count only in the “all” columns, where a few participants above 1.01 do not sink an otherwise sound fit. The verdict is the worst of the coloured cells. An acceptable fit can be interpreted at the population level. The ones here all have their 8 chains in one mode. They miss the strictest cut-offs by about what the maximum over some 40 parameters gives by chance, or by a handful of divergent draws out of 8,000. The fits that did not converge miss by far more, with R-hat around 1.5 or above.

# What compute_convergence() in server/predictions.R stored for each model. A
# prediction file made before it existed has none, and is listed as not computed.
show_convergence <- function(preds) {
  conv <- lapply(preds, \(p) p$convergence)
  have <- !vapply(conv, is.null, logical(1))
  if (any(!have)) {
    cat("Not computed yet (re-run `./hpc predict` and pull): ",
        paste(names(preds)[!have], collapse = ", "), "
", sep = "")
  }
  if (!any(have)) return(invisible(NULL))
  conv <- conv[have]
  sampler <- vapply(preds[have], \(p) p$meta$sampler, character(1))
  get <- \(k) vapply(conv, \(x) as.numeric(x[[k]]), numeric(1))

  rhat_pop <- get("rhat_pop_max")
  rhat_all <- get("rhat_all_max")
  rhat_lp <- get("rhat_lp")
  ess <- get("ess_bulk_pop_min")
  div <- get("divergent")
  div_pct <- get("divergent_pct")

  # 1 = satisfactory, 2 = acceptable, 3 = poor. A missing value (the sampler
  # diagnostics could not be read) is not counted against the model.
  grade <- \(x, ok, warn, higher_is_worse = TRUE) {
    if (!higher_is_worse) { x <- -x; ok <- -ok; warn <- -warn }
    ifelse(is.na(x), 1L, ifelse(x <= ok, 1L, ifelse(x <= warn, 2L, 3L)))
  }
  g_rhat <- grade(rhat_pop, 1.01, 1.05)
  g_lp <- grade(rhat_lp, 1.01, 1.1)
  g_ess <- grade(ess, 400, 100, higher_is_worse = FALSE)
  g_div <- ifelse(is.na(div), 1L, ifelse(div == 0, 1L, ifelse(div_pct < 1, 2L, 3L)))
  g_all <- pmax(g_rhat, g_lp, g_ess, g_div)

  palette <- c("#C8E6C9", "#FFE082", "#EF9A9A")
  verdict <- c("Satisfactory", "Acceptable", "Poor")

  n_params <- get("n_params")
  tab <- data.frame(
    Model = names(conv),
    Sampler = ifelse(sampler == "stanli", "stanli (stand-in)", sampler),
    Chains = get("n_chains"),
    "R-hat, population (max)" = formatC(rhat_pop, format = "f", digits = 3),
    "R-hat, all (max)" = formatC(rhat_all, format = "f", digits = 3),
    "All parameters above 1.01" = sprintf("%.1f%%", 100 * get("n_rhat_101_all") / n_params),
    "R-hat, lp__" = formatC(rhat_lp, format = "f", digits = 3),
    "Bulk ESS, population (min)" = formatC(ess, format = "f", digits = 0, big.mark = ","),
    "Divergent" = ifelse(is.na(div), "n/a", sprintf("%d (%.2f%%)", as.integer(div), div_pct)),
    "At max treedepth" = ifelse(is.na(get("treedepth_hit_pct")), "n/a",
                                sprintf("%.0f%%", get("treedepth_hit_pct"))),
    Verdict = verdict[g_all],
    check.names = FALSE
  )

  out <- tinytable::tt(tab) |>
    tinytable::style_tt(j = 3:10, align = "r") |>
    fill_cells(1, model_fill(tab$Model)) |>
    fill_cells(4, palette[g_rhat]) |>
    fill_cells(7, palette[g_lp]) |>
    fill_cells(8, palette[g_ess]) |>
    fill_cells(9, palette[g_div]) |>
    fill_cells(11, palette[g_all])
  out
}

Participants That Do Not Mix

Even in the models that converge, a few dozen participants do not mix. Their own parameters sit in one of two local solutions, and no chain switches between them. Almost always it is the non-decision time traded against the boundary or the outlier probability (server/AGENT.md 4.10). None is excluded: no rule on the data picks them out without also removing many participants who mix well. The list comes from server/mixing_participants.R.

# Total compute, three ways, for a rough headline number:
# - Wall clock: the shards of a model run at the same time on the cluster, so
#   what one waited for is the longest chain (queueing, compilation and the
#   combine/loo step are not included).
# - Chains one after the other: the sum of the chain times, i.e. the same run
#   with no parallel chains but still multithreaded within each chain.
# - Single core: that sum times the threads per chain. This assumes within-
#   chain threading scales linearly, which is close to what was measured
#   (4 -> 8 threads halved the time per chain, server/AGENT.md 4.4.1), so read
#   it as an order of magnitude, not a measurement.
table_compute_time <- function(preds) {
  tidy_efficiency(preds) |>
    summarise(.by = Model,
              Chains = n(),
              Threads = first(Threads),
              Cores = n() * first(Threads),
              Wall = max(Time_total) / 3600,
              Serial = sum(Time_total) / 3600,
              SingleCore = sum(Time_total * Threads) / 3600) |>
    mutate(SingleCore_days = SingleCore / 24,
           SingleCore_years = SingleCore / (24 * 365.25)) |>
    gt::gt(rowname_col = "Model") |>
    gt::cols_label(
      Chains = "Chains",
      Threads = "Threads / chain",
      Cores = "Cores used",
      Wall = "Wall clock (h)",
      Serial = "Chains in sequence (h)",
      SingleCore = "Single core (h)",
      SingleCore_days = "Single core (days)",
      SingleCore_years = "Single core (years)"
    ) |>
    gt::fmt_number(columns = c(Wall, Serial, SingleCore, SingleCore_days), decimals = 0) |>
    gt::fmt_number(columns = SingleCore_years, decimals = 2) |>
    gt::tab_spanner(label = "Without parallelisation (estimated)",
                    columns = c(Serial, SingleCore, SingleCore_days, SingleCore_years)) |>
    gt::tab_source_note("Warmup + sampling only. Single core assumes linear scaling of within-chain threading.")
}

# loo was added at combine time and lives in the fit's $criteria$loo, which the
# prediction files carry over. The list names become the model labels.
# loo_compare() needs two models at least, which an illusion only just started
# on does not have yet.
show_loo <- function(preds) {
  if (length(preds) < 2) {
    cat("Only one model so far (", names(preds), "): nothing to compare.\n", sep = "")
    return(invisible(NULL))
  }
  rez_loo <- loo_compare(lapply(preds, \(p) p$loo))
  print(report(rez_loo))

  # Ranked by ELPD, best first, as loo_compare() has them. LOOIC is left out: it
  # is -2 x ELPD, so it ranks the models the same way but in the opposite
  # direction (lower is better), which only confuses the table. The model names
  # carry their colour from the plots, and the ELPD cells are shaded from the
  # worst model to the best, so the gaps can be read off the colour. The
  # difference is given with its SE, and the best model, which has none, with a
  # dash. p is the z-test of the difference to the best model.
  tab <- as.data.frame(parameters(rez_loo))
  num <- \(x) formatC(x, format = "f", digits = 1, big.mark = ",")
  best <- seq_len(nrow(tab)) == 1
  weight <- if (diff(range(tab$ELPD)) == 0) 1 else (tab$ELPD - min(tab$ELPD)) / diff(range(tab$ELPD))
  elpd_fill <- rgb(colorRamp(c("#f7fcf5", "#238b45"))(weight), maxColorValue = 255)

  # Formatted by hand (one decimal, thousands marks): format_tt()'s digits are
  # significant digits, which writes 0 and -176 next to -2,372.7
  tab <- data.frame(
    Model = tab$Name,
    ELPD = num(tab$ELPD),
    "ELPD diff ± SE" = ifelse(best, "—", paste0(num(tab$Difference), " ± ", num(tab$Difference_SE))),
    p = ifelse(is.na(tab$p), "", insight::format_p(tab$p, name = NULL)),
    check.names = FALSE
  )

  out <- tinytable::tt(tab) |>
    tinytable::style_tt(j = 2:4, align = "r") |>
    fill_cells(1, model_fill(tab$Model)) |>
    fill_cells(2, elpd_fill) |>
    tinytable::style_tt(i = 1, j = 2, bold = TRUE)
  out
}

# What the fit costs against how well it does: the elpd against the time a chain
# took, the y axis of the computational cost plot. The elpd is relative to the
# best model (loo_compare), with the SE of that difference: each model's own SE
# (~800 here) is mostly shared between models fitted to the same trials, and
# would swamp differences that are in fact well resolved.
plot_loo_cost <- function(preds) {
  if (length(preds) < 2) {
    cat("Only one model so far (", names(preds), "): nothing to compare.\n", sep = "")
    return(invisible(NULL))
  }
  # loo 2.10 returns a data frame with the names in a `model` column, earlier
  # versions a matrix with them as row names
  rez_loo <- as.data.frame(loo_compare(lapply(preds, \(p) p$loo)))
  elpd <- data.frame(
    Model = factor(if (is.null(rez_loo$model)) rownames(rez_loo) else rez_loo$model,
                   levels = names(preds)),
    elpd_diff = rez_loo$elpd_diff,
    se_diff = rez_loo$se_diff
  )

  # One point per chain, as in the cost plot: a model has one elpd, so its
  # chains spread along x only
  efficiency <- tidy_efficiency(preds) |>
    mutate(Time_total = Time_total / 3600) |>
    left_join(elpd, by = "Model")
  efficiency_means <- efficiency |>
    summarise(.by = c(Model, elpd_diff, se_diff),
              Time_total = mean(Time_total))

  ggplot(efficiency, aes(x = Time_total, y = elpd_diff, colour = Model)) +
    geom_hline(yintercept = 0, colour = "grey60", linetype = "dashed") +
    geom_point(size = 2.5, alpha = 0.35) +

# All models are checked against the same trials: the observed data is taken
# from the first model's file, so the models must share an illusion. What is
# predicted, and how, is compute_ppc() in server/predictions.R.
plot_ppc <- function(preds, rt_range = c(0, 2.5), binwidth = 0.02, n_points = 512, linewidth = 0.9) {

  if (is.null(names(preds))) names(preds) <- paste("Model", seq_along(preds))
  if (length(unique(vapply(preds, \(p) p$meta$illusion, character(1)))) > 1) {
    stop("plot_ppc() compares models fitted to the same illusion.")
  }

  # Observed: histogram of every trial, put on the same defective-density
  # scale. Stored as the count of each distinct (RT, Error), so this is exact.
  observed <- preds[[1]]$ppc$observed
  obs_data <- observed |>
    mutate(bin = floor(RT / binwidth)) |>
    count(Error, bin, wt = n, name = "n") |>
    mutate(
      x = (bin + 0.5) * binwidth,
      y = n / (sum(observed$n) * binwidth) * if_else(Error == 0, 1, -1),
      Response = factor(Error, levels = c(0, 1))
    ) |>
    filter(x <= max(rt_range))

  # Predicted: defective densities, each response's curve scaled by its own
  # rate, so the two areas sum to 1 and correct/error are directly comparable
  # across models. The predicted RTs come binned (1 ms) with the bandwidth
  # density() picked on the raw values, so this is the curve the raw RTs give,
  # to within a bin: density() itself bins its input far more coarsely.
  defective_density <- function(pp) {
    total <- sum(pp$n)
    purrr::map_dfr(c(0, 1), function(r) {
      r_chr <- as.character(r)
      if (pp$n[[r_chr]] < 2) return(NULL)
      d <- pp$counts[pp$counts$Response == r, ]
      dens <- density((d$Bin + 0.5) * pp$bin, weights = d$n / sum(d$n), bw = pp$bw[[r_chr]],
                      from = rt_range[1], to = rt_range[2], n = n_points)
      data.frame(
        Response = factor(r, levels = c(0, 1)),
        x = dens$x,
        y = dens$y * (pp$n[[r_chr]] / total) * if (r == 0) 1 else -1
      )
    })
  }
  pred_data <- purrr::imap_dfr(preds, \(p, nm) mutate(defective_density(p$ppc$predicted), Model = nm))
  pred_data$Model <- factor(pred_data$Model, levels = names(preds))

  ggplot(mapping = aes(x = x, y = y)) +
    geom_col(
      data = obs_data,
      aes(fill = Response),
      width = binwidth, alpha = 0.4
    ) +
    geom_line(
      data = pred_data,
      aes(color = Model, group = interaction(Model, Response)),
      linewidth = linewidth, alpha = 0.9
    ) +
    geom_hline(yintercept = 0, color = "grey30", linewidth = 0.3) +
    scale_fill_manual(
      values = c("0" = "#43A047", "1" = "#F44336"),
      labels = c("0" = "Correct", "1" = "Error")
    ) +
    scale_color_manual(values = model_colors) +
    scale_y_continuous(labels = abs) +
    coord_cartesian(xlim = rt_range) +
    labs(x = "RT", y = "Density", fill = "Observed", color = "Predicted") +
    guides(fill = guide_legend(override.aes = list(alpha = 0.6))) +
    theme_minimal() +
    theme(legend.position = "bottom")
}

# Display names for the parameters, in the order they are plotted. Names are
# the family's own ("mu", "ndt", ...), values are what goes in the panel
# titles. Several parameters mean different things from one family to the next
# - "mu" is a drift rate in the DDM and the log-mean of the correct response in
# the LNR, and "sigmabias" is the variability of the starting point in the DDM
# but its maximum (A) in the racing models - so the map is built per family
# (the brms family name, e.g. "cogmod_ddm", as stored in the prediction files).
param_labels <- function(family) {
  lvls <- c("Error Rate" = "Error Rate",
            "Mean RT" = "Mean RT",
            "Median RT" = "Median RT",
            "mu" = "mu",
            "nuone" = "Nu (Errors)",
            "driftone" = "Drift (Errors)",
            "sigmadrift" = "Drift Variability",
            "sigmazero" = "Sigma (Correct)",
            "sigmaone" = "Sigma (Errors)",
            "boundary" = "Boundary",
            "bias" = "Bias",
            "sigmabias" = "Bias Variability",
            "ndt" = "NDT",
            "sigmandt" = "NDT Variability")
  switch(
    family,
    "cogmod_ddm" = {
      lvls["mu"] <- "Drift"
    },
    # sigmabias is only estimated in LNR-6, where it is the start-point range
    # of the race, as in the other racing families. Each accumulator's
    # parameters are kept together - the correct response's nu and its
    # variability, then the error's - rather than the two nus first.
    "cogmod_lnr" = {
      lvls[c("mu", "sigmazero", "sigmaone", "sigmabias")] <-
        c("Nu (Correct)", "Nu Variability (Correct)", "Nu Variability (Errors)",
          "Start Point Range")
      first <- c("Error Rate", "Mean RT", "Median RT", "mu", "sigmazero", "nuone", "sigmaone")
      lvls <- lvls[c(first, setdiff(names(lvls), first))]
    },
    # Both racing families: mu is the drift of the accumulator for correct
    # responses and driftone that of the other one, and sigmabias is the
    # start-point range A rather than a variability around a bias
    "cogmod_rdm" = {
      lvls[c("mu", "sigmabias")] <- c("Drift (Correct)", "Start Point Range")
    },
    "cogmod_lba2" = {
      lvls[c("mu", "sigmabias", "sigmazero", "sigmaone")] <-
        c("Drift (Correct)", "Start Point Range",
          "Drift SD (Correct)", "Drift SD (Errors)")
    }
  )
  lvls
}

# Rename the parameters and put them in canonical order. Parameter_Order keeps
# that order recoverable once the models are stacked together, where the factor
# levels of families that do not share their parameters get merged in whatever
# order the models happen to come in. It is the position in the full map rather
# than among the parameters this model has, so that it is comparable between
# models of a family that do not estimate the same ones.
format_parameters <- function(pred, family) {
  lvls <- param_labels(family)
  found <- unique(as.character(pred$Parameter))
  # A parameter the map does not know about (a family added later) keeps its
  # own name and goes last, rather than silently becoming an NA panel
  unknown <- setdiff(found, names(lvls))
  if (length(unknown) > 0) lvls <- c(lvls, setNames(unknown, unknown))
  pred$Parameter_Order <- match(as.character(pred$Parameter), names(lvls))
  lvls <- lvls[names(lvls) %in% found]

  pred$Parameter <- factor(unname(lvls[as.character(pred$Parameter)]),
                           levels = unname(lvls))
  pred
}

# The predictions of each of an illusion's models, relabelled, back in raw
# units, and stacked. Adding a model to `models` in the loading chunk is all it
# takes for it to get its own column in every plot below.
tidy_predictions <- function(preds, what) {
  purrr::imap_dfr(preds, \(p, name) {
    p[[what]] |>
      format_parameters(p$meta$family) |>
      # Restore the original scale of the illusion strength and difference
      restore_units(illusion = p$meta$illusion) |>
      mutate(Model = name, Family = p$meta$family)
  }) |>
    mutate(Model = factor(Model, levels = names(preds)))
}

# Height of the column-header row, in units of one panel row. A figure that
# puts something alongside the grid has to skip the headers to line up with the
# panels, so this is a shared constant rather than a default buried in the
# function below.
parameter_grid_header <- 0.35

# Which parameter goes in which row of which column. Models of one family share
# their parameters, so within a family rows are matched by name: the family's
# rows are every parameter any of its models has, in canonical order, and a
# model that does not estimate one (DDM-4's drift variability, say) gets a gap
# there. Families do not share them, so across families rows cannot be matched
# by name: each family's rows are packed from the top (which lines up the drift
# rates and the nus, both "mu"), and the non-decision time - the one parameter
# every family has - is anchored to the bottom row. The exception is a
# parameter that means the same thing in several families (`align`, the
# start-point range of the racing models): it goes in the same row everywhere,
# the lowest it falls in any family, with gaps above it in the others. Shorter
# columns get NA in between, drawn as spacers, so the columns stay aligned
# however many parameters they have.
parameter_grid_rows <- function(data, anchor_bottom = c("NDT", "NDT Variability"),
                                align = "Start Point Range") {
  data$Model <- droplevels(factor(data$Model))
  # Data without a family (made by hand, say) treats each model as its own
  if (is.null(data$Family)) data$Family <- as.character(data$Model)

  in_order <- function(d) unique(as.character(d$Parameter[order(d$Parameter_Order)]))
  families <- lapply(split(data, data$Family), \(d) {
    rows <- in_order(d)
    rows[!rows %in% anchor_bottom]
  })
  # Pad the families' top rows so that each aligned parameter lands in the row
  # of the family where it comes lowest
  for (p in align) {
    target <- max(0, unlist(lapply(families, \(rows) match(p, rows))), na.rm = TRUE)
    families <- lapply(families, \(rows) {
      at <- match(p, rows)
      if (is.na(at) || at == target) return(rows)
      append(rows, rep(NA_character_, target - at), after = at - 1)
    })
  }

  slots <- lapply(levels(data$Model), function(model) {
    family <- data$Family[data$Model == model][1]
    family_rows <- in_order(data[data$Family == family, ])
    own <- in_order(data[data$Model == model, ])
    top <- families[[family]]
    bottom <- family_rows[family_rows %in% anchor_bottom]
    list(top = ifelse(top %in% own, top, NA_character_),
         bottom = ifelse(bottom %in% own, bottom, NA_character_))
  })
  names(slots) <- levels(data$Model)
  n_top <- max(lengths(lapply(slots, `[[`, "top")))
  n_bottom <- max(lengths(lapply(slots, `[[`, "bottom")))

  lapply(slots, function(s) {
    c(s$top, rep(NA_character_, n_top - length(s$top)),
      rep(NA_character_, n_bottom - length(s$bottom)), s$bottom)
  })
}

# One column per model, one row per parameter, with panel(model, parameter)
# drawing each cell. The cells go into a single grid, filled column by column,
# rather than one sub-layout per model put side by side: patchwork only aligns
# panels within a layout, so separate columns let each one's rows drift with
# its own titles and legends.
plot_parameter_grid <- function(data, panel,
                                anchor_bottom = c("NDT", "NDT Variability"),
                                header_height = parameter_grid_header) {
  rows_by_model <- parameter_grid_rows(data, anchor_bottom)
  cells <- unlist(Map(function(model, rows) {
    c(list(make_header(model)),
      lapply(rows, function(p) if (is.na(p)) plot_spacer() else panel(model, p)))
  }, names(rows_by_model), rows_by_model), recursive = FALSE)

  wrap_plots(cells, ncol = length(rows_by_model), byrow = FALSE,
             heights = c(header_height, rep(1, length(rows_by_model[[1]]))))
}

# outline, which is in its raw units.
make_heatmap <- function(preds_heatmaps, model, parameter, illusion) {
  d <- filter(preds_heatmaps, Model == model, Parameter == parameter)
  region <- linear_regions[[illusion]]
  contours <- if (heatmap_labels) {
    list(
      # Few enough lines that the labels have room: 20 would overlap in panels
      # this short. Breaks strictly inside the range, as pretty() can overshoot.
      geomtextpath::geom_textcontour(
        # The label is given explicitly: the default rounds it, which writes
        # 0.4 on every NDT contour from 0.37 to 0.41. The sign is a true minus
        # (U+2212): geomtextpath 0.2.0 silently drops an ASCII hyphen, which
        # labels -3 as 3 and makes the colour scale look reversed.
        aes(z = Predicted,
            label = after_stat(sub("^-", "−", format(level, trim = TRUE, drop0trailing = TRUE)))),
        breaks = (\(b) b[b > min(d$Predicted) & b < max(d$Predicted)])(pretty(d$Predicted, n = 10)),
        size = 2.5, linecolour = "white", textcolour = "white", linewidth = 0.4, alpha = 0.5,
        padding = grid::unit(0.5, "mm"), straight = TRUE
      ),
      guides(fill = "none")
    )
  } else {
    geom_contour(aes(z = Predicted), color = "white", alpha = 0.3, linewidth = 0.5, bins = 20)
  }

  ggplot(d, aes(x = Illusion_Strength, y = Illusion_Difference)) +
    geom_tile(aes(fill = Predicted)) +
    contours +
    geom_vline(xintercept = 0, color = "white", linewidth = 0.3, linetype = "dashed") +

    interpolate = TRUE
  ))
}

# The colour and the transparency of each task difficulty level (`levels`,
# sorted), each placed by its value on one gradient. Its red to green middle
# runs over the difficulty range of the linear-approximation region
# (linear_regions, with the heatmaps), red at the hard edge and green at the
# easy one. Outside it the gradient carries on, faded: from the red through
# pink to purple at the hardest level, and from the green through teal to cyan
# at the easiest. An illusion with no region has its whole range on the middle.
difficulty_scale <- function(levels, region = NULL) {
  bounds <- if (is.null(region)) range(levels) else region$difference
  # Tolerance for levels on the bounds, which carry floating-point error
  below <- levels < bounds[1] - 1e-8
  above <- levels > bounds[2] + 1e-8
  inside <- !below & !above

  middle <- grDevices::colorRamp(c("#FF5722", "#FFC107", "#8BC34A", "#4CAF50"))
  harder <- grDevices::colorRamp(c("#FF5722", "#E91E63", "#9C27B0"))
  easier <- grDevices::colorRamp(c("#4CAF50", "#009688", "#00BCD4"))
  hex <- \(m) grDevices::rgb(m, maxColorValue = 255)

  cols <- hex(middle((pmin(pmax(levels, bounds[1]), bounds[2]) - bounds[1]) / diff(bounds)))
  if (any(below)) {
    cols[below] <- hex(harder((bounds[1] - levels[below]) / (bounds[1] - min(levels))))
  }
  if (any(above)) {
    cols[above] <- hex(easier((levels[above] - bounds[2]) / (max(levels) - bounds[2])))
  }
  list(colour = setNames(cols, levels),
       alpha = setNames(ifelse(inside, 1, 0.5), levels),
       inside = setNames(inside, levels))
}

# One element of preds_curves, i.e. one illusion, whose name is passed along
# for its linear-approximation region. A prediction file made before
# 2026-11-16 has no median RT and 5 difficulty levels: it shows the mean RT
# instead, at those 5 levels. Its chunks have warning: false for the RT, which
# is NA where a level predicts (almost) no correct responses, and ggplot says
# so once per layer.
plot_best_model <- function(preds_curves, illusion, model = "DDM-5") {
  # A model left out of an illusion's list (not converged, still fitting) has
  # no curves there
  if (!model %in% preds_curves$Model) {
    return(cat(model, "is not among this illusion's models.\n"))
  }
  d <- preds_curves |>
    filter(Model == model) |>
    mutate(Illusion_Difference = round(Illusion_Difference, 5))
  rt <- if ("Median RT" %in% d$Parameter) "Median RT" else "Mean RT"
  d <- d |>
    filter(!Parameter %in% setdiff(c("Mean RT", "Median RT"), rt)) |>
    # Panels in canonical order, the two outcomes first
    arrange(Parameter_Order) |>
    mutate(Parameter = factor(as.character(Parameter), levels = unique(as.character(Parameter))))

  # Left column: the error rate above the RT, half the height each. Right
  # column: one row per parameter, however many the model has.
  n_par <- nlevels(d$Parameter) - 2
  design <- matrix(c(rep(1:2, each = n_par), rep(seq_len(n_par) + 2, each = 2)), ncol = 2)

  region <- linear_regions[[illusion]]
  lv <- sort(unique(d$Illusion_Difference))
  sc <- difficulty_scale(lv, region)
  d$Inside <- sc$inside[as.character(d$Illusion_Difference)]
  d$Illusion_Difference <- factor(d$Illusion_Difference, levels = lv)

  # Two layers, so that the faded lines are drawn under the others rather than
  # over them. The colour and alpha scales share their title and levels, so
  # ggplot merges them into one legend, with faded keys. The breaks are given,
  # or the legend lists the levels in the order the layers bring them in, and
  # the labels rounded (0.06375 otherwise, on the Vertical-Horizontal).
  level_labels <- \(x) signif(as.numeric(x), 2)
  # The strengths outside the region, shaded: the curves only show its
  # difficulty bounds otherwise
  outside <- if (!is.null(region)) {
    data.frame(xmin = c(-Inf, region$strength[2]), xmax = c(region$strength[1], Inf)) |>
      filter(xmax > min(d$Illusion_Strength), xmin < max(d$Illusion_Strength))
  }
  ggplot(d, aes(Illusion_Strength, Predicted, color = Illusion_Difference,
                alpha = Illusion_Difference, group = Illusion_Difference)) +
    (if (!is.null(outside) && nrow(outside) > 0) {
      geom_rect(data = outside, aes(xmin = xmin, xmax = xmax, ymin = -Inf, ymax = Inf),
                inherit.aes = FALSE, fill = "grey50", alpha = 0.12)
    }) +
    geom_line(data = \(x) filter(x, !Inside), linewidth = 1.1) +
    geom_line(data = \(x) filter(x, Inside), linewidth = 1.1) +
    scale_color_manual(values = sc$colour, breaks = names(sc$colour), labels = level_labels) +
    scale_alpha_manual(values = sc$alpha, breaks = names(sc$alpha), labels = level_labels) +
    ggh4x::facet_manual(vars(Parameter), design = design, scales = "free_y") +
    labs(title = model,
         subtitle = if (!is.null(region)) {
           paste0("Linear-approximation region: task difficulty ", region$difference[1],
                  " to ", region$difference[2], ", illusion strength ", region$strength[1],
                  " to ", region$strength[2], " (shaded outside)")
         },
         x = "Illusion Strength", y = NULL,
         color = "Task Difficulty", alpha = "Task Difficulty") +
    theme_minimal()
}

# What each model estimates, read from its prediction files: the parameters it
# has curves for are the ones given a smooth
model_table <- function(preds) {
  fits <- unlist(unname(preds), recursive = FALSE)
  fits <- fits[!duplicated(names(fits))]
  purrr::imap_dfr(fits[paper_models], \(p, model) {
    map <- family_components[[p$meta$family]]
    free <- intersect(unique(p$curves$Parameter), names(map))
    components |>
      mutate(Model = model, Family = p$meta$family,
             N_family = vapply(Component, \(k) sum(map == k), numeric(1)),
             N_free = vapply(Component, \(k) sum(map[free] == k), numeric(1)),
             N_total = length(free))
  }) |>
    mutate(State = case_when(N_free > 0 ~ "free", N_family > 0 ~ "fixed", .default = "absent"),
           Text = case_when(State == "free" & N_family > 1 & N_free == N_family ~ "one per\naccumulator",
                            State == "free" & N_free < N_family ~ "error\naccumulator",
                            State == "free" ~ "✓", State == "fixed" ~ "fixed", .default = "–"))
}

# Panels A and B, the two architectures. The paths are simulated with fixed
# seeds, picked for paths that are easy to read, so the figure does not change
# from one render to the next.
simulate_path <- function(drift, start, bound, lower = -Inf, noise = 0.32, seed = 1, t0 = 0.4,
                          dt = 0.004, n = 600) {
  set.seed(seed)
  x <- start + cumsum(drift * dt + noise * sqrt(dt) * rnorm(n))
  hit <- which(x >= bound | x <= lower)[1]
  if (is.na(hit)) hit <- n
  x <- pmin(pmax(x[seq_len(hit)], lower), bound)
  data.frame(Time = t0 + c(0, seq_len(hit) * dt), Evidence = c(start, x))
}

# The posterior predictive check of the three illusions side by side, with one
# legend. plot_ppc() is the one of the section above. The legend of the models
# is taken from the panel that has the most of them: patchwork only merges
# legends that are identical, and an illusion with a model missing has another.
#
# The models differ most at the peak of the correct responses, where their
# lines are on top of each other on a plain axis. Two ways of pulling them
# apart:
# - `y_scale`: "square" stretches the high densities (and flattens the errors
#   and the tails), "sqrt" does the opposite, "linear" leaves the axis alone.
#   The transform is of the size of the density, so that errors stay mirrored.
# - `zoom`: two insets, with the peak of the correct responses and that of the
#   errors enlarged, and a frame around what each enlarges.
signed_power <- function(power) {
  scales::new_transform(paste0("signed_power_", power),
                        transform = \(x) sign(x) * abs(x)^power,
                        inverse = \(x) sign(x) * abs(x)^(1 / power))
}
plot_ppc_paper <- function(preds, y_scale = c("linear", "square", "sqrt"), zoom = FALSE) {
  y_scale <- match.arg(y_scale)
  preds <- lapply(preds, \(p) p[intersect(paper_models, names(p))])
  with_legend <- which.max(lengths(preds))
  panels <- lapply(seq_along(preds), \(i) {
    p <- plot_ppc(preds[[i]], rt_range = c(0, 2), linewidth = 0.45) +
      labs(title = paper_illusions[[names(preds)[i]]], x = "Response time (s)",
           y = if (i == 1) "Density" else NULL) +
      guides(fill = guide_legend(order = 1, override.aes = list(alpha = 0.6)),
             color = if (i == with_legend) {
               guide_legend(order = 2, nrow = 1, override.aes = list(linewidth = 1))
             } else {
               "none"
             }) +
      theme_minimal(base_size = 9) +
      theme(plot.title = element_text(face = "bold", size = 9.5, hjust = 0.5),
            panel.grid.minor = element_blank(),
            panel.grid.major = element_line(linewidth = 0.25))
    if (y_scale != "linear") {
      p <- suppressMessages(
        p + scale_y_continuous(transform = signed_power(c(square = 2, sqrt = 0.5)[[y_scale]]),
                               breaks = c(-0.5, -0.1, 0, 0.1, 0.5, 1, 1.5, 2), labels = abs)
      )
    }
    if (!zoom) return(p)

    # An inset is the same plot cut to a window, on a plain axis
    observed <- p$layers[[1]]$data
    predicted <- p$layers[[2]]$data
    top <- max(observed$y, predicted$y)
    inset <- function(window_x, window_y) {
      ggplotGrob(suppressMessages(
        p + scale_y_continuous(labels = abs) +
          coord_cartesian(xlim = window_x, ylim = range(window_y), expand = FALSE) +
          labs(title = NULL, x = NULL, y = NULL) +
          theme(legend.position = "none", axis.text = element_blank(),
                panel.grid.major = element_blank(), panel.grid.minor = element_blank(),
                panel.border = element_rect(fill = NA, colour = "grey30", linewidth = 0.4),
                plot.background = element_rect(fill = "white", colour = NA),
                plot.margin = margin(0, 0, 0, 0))
      ))
    }
    frame <- function(window_x, window_y) {
      annotate("rect", xmin = window_x[1], xmax = window_x[2], ymin = min(window_y), ymax = max(window_y),
               fill = NA, colour = "grey30", linewidth = 0.3)
    }

    # The peak of the correct responses: the observed mode, 0.12 s either
    # side, and the top quarter of the densities there. Its inset is in the
    # empty corner above the tail.
    correct_x <- observed$x[which.max(observed$y)] + c(-0.12, 0.12)
    correct_y <- c(0.72, 1.03) * top

    # The peak of the errors, which is wider and where the models differ more
    # in proportion: 0.15 s either side of where the models put it (the
    # observed histogram of the errors is too rough to give a mode), and the
    # lower half of the densities there. Its inset is under the axis, which is
    # extended to make room for it.
    errors <- filter(predicted, y < 0)
    error_mode <- errors |>
      summarise(x = x[which.min(y)], .by = Model) |>
      pull(x) |>
      median()
    error_x <- error_mode + c(-0.15, 0.15)
    depth <- min(errors$y, observed$y[observed$x > error_x[1] & observed$x < error_x[2]])
    error_y <- c(0.5, 1.04) * depth

    p +
      expand_limits(y = -0.59 * top) +
      frame(correct_x, correct_y) +
      frame(error_x, error_y) +
      annotation_custom(inset(correct_x, correct_y), xmin = 0.95, xmax = 2.05, ymin = 0.42 * top, ymax = 1.03 * top) +
      annotation_custom(inset(error_x, error_y), xmin = 0.95, xmax = 2.05, ymin = -0.58 * top, ymax = -0.13 * top)
  })
  wrap_plots(panels, nrow = 1, guides = "collect") &
    theme(legend.position = "bottom", legend.box = "vertical", legend.margin = margin(t = -2),
          legend.spacing.y = grid::unit(0, "pt"), legend.key.size = grid::unit(10, "pt"),
          legend.text = element_text(size = 8), legend.title = element_text(size = 8))
}

# One row per fit, with everything the paper reports about it. The ELPD
# difference is from the retained model on the same illusion, with the SE of
# the difference from the paired trial-wise values, as loo_compare() gives it.
summarise_fits <- function(preds, reference = retained_model) {
  purrr::imap_dfr(preds, \(fits, illusion) {
    ref <- fits[[reference]]$loo$pointwise[, "elpd_loo"]
    purrr::imap_dfr(fits, \(p, model) {
      cv <- p$convergence
      d <- p$loo$pointwise[, "elpd_loo"] - ref
      data.frame(
        Illusion = illusion, Model = model, Sampler = p$meta$sampler,
        Rhat = cv$rhat_pop_max, Rhat_lp = cv$rhat_lp, ESS = cv$ess_bulk_pop_min,
        Divergent_pct = cv$divergent_pct,
        ELPD = p$loo$estimates["elpd_loo", "Estimate"], ELPD_SE = p$loo$estimates["elpd_loo", "SE"],
        ELPD_diff = sum(d), ELPD_diff_SE = sqrt(length(d)) * sd(d),
        Pareto_k_pct = 100 * mean(p$loo$diagnostics$pareto_k > 0.7),
        # Compute: how long a chain took on average, and the total over the
        # chains and their threads, which is what one core would have taken
        # (assuming threading scales linearly, server/AGENT.md 4.4.1)
        Chain_days = mean(p$efficiency$Time_total) / 86400,
        Core_days = sum(p$efficiency$Time_total * p$efficiency$Threads) / 86400,
        Core_hours = sum(p$efficiency$Time_total * p$efficiency$Threads) / 3600
      )
    })
  }) |>
    # The criteria of the paper (Evaluation), which are the "acceptable" ones
    # of the convergence tables above
    mutate(Converged = Rhat <= 1.05 & Rhat_lp <= 1.1 & ESS >= 100 & Divergent_pct < 1)
}

# Axis breaks strictly inside the range, so that the labels of neighbouring

  \(limits) {

  }
}

Only the sampler differs: the same Stan program, data, priors, inits, shards, chains and threads. Two samplers of one posterior should agree on everything but speed, to within Monte Carlo error. A larger difference means that one of them missed part of the posterior. Convergence then says which one to trust, not the sampler.

# One row per fit, the two fits of a model one above the other. The ELPD
# difference is stanli's minus cmdstanr's, with the SE that loo_compare() would
# give it: both fits are of the same trials, so the pointwise values pair up
# (`yhash`, loo's hash of the outcome, checks that they do). The Monte Carlo
# SE is that of the difference, from every trial's own MCSE, and is the yardstick:
# a difference within about twice it is what two runs of one sampler would give.
# loo's mcse_loo() returns NA as soon as one trial has a Pareto k above 0.7,
# which every fit here has, and those trials' MCSEs are underestimated, so read
# it as a lower bound.
table_sampler_agreement <- function(twin_preds) {
  rows <- purrr::imap_dfr(twin_preds, \(pair, label) {
    loo_c <- pair$cmdstanr$loo
    loo_s <- pair$stanli$loo
    if (!identical(attr(loo_c, "yhash"), attr(loo_s, "yhash"))) {
      stop(label, ": the two fits are not of the same trials.")
    }
    d <- loo_s$pointwise[, "elpd_loo"] - loo_c$pointwise[, "elpd_loo"]
    mcse <- sqrt(sum(loo_c$pointwise[, "mcse_elpd_loo"]^2, loo_s$pointwise[, "mcse_elpd_loo"]^2))
    purrr::imap_dfr(pair, \(p, sampler) {
      conv <- p$convergence
      data.frame(
        Model = label,
        Family = sub(" .*", "", label),
        Sampler = sampler,
        ELPD = sprintf("%s ± %s", formatC(p$loo$estimates["elpd_loo", 1], format = "f", digits = 1, big.mark = ","),
                       formatC(p$loo$estimates["elpd_loo", 2], format = "f", digits = 1, big.mark = ",")),
        p_loo = formatC(p$loo$estimates["p_loo", 1], format = "f", digits = 1, big.mark = ","),
        "ELPD diff ± SE" = if (sampler == "stanli") sprintf("%.1f ± %.1f", sum(d), sqrt(length(d)) * sd(d)) else "",
        "Monte Carlo SE" = if (sampler == "stanli") sprintf("%.1f", mcse) else "",
        "R-hat, population (max)" = formatC(conv$rhat_pop_max, format = "f", digits = 3),
        "R-hat, lp__" = formatC(conv$rhat_lp, format = "f", digits = 3),
        # Chains in different modes show here first: within a mode the chain
        # means of lp__ sit within a few tens of each other (sd ~110)
        "lp__, spread of the chain means" = formatC(diff(range(conv$lp_chain_mean)), format = "f", digits = 0, big.mark = ","),
        check.names = FALSE
      )
    })
  })
  tinytable::tt(select(rows, -Family)) |>
    tinytable::style_tt(j = 3:9, align = "r") |>
    fill_cells(1, model_fill(rows$Family))
}

plot_sampler_curves <- function(pair, label) {
  # The mean RT stands for the RT (the median has no SE to measure the gap
  # in), and the 5 difficulty levels every file has (coarse_levels()). The Z
  # difference is rounded so that the same level from a 5-level and a 9-level
  # file pairs up.
  curves <- purrr::imap_dfr(pair, \(p, sampler) mutate(p$curves, Sampler = sampler)) |>
    restore_units(illusion = pair$stanli$meta$illusion) |>
    filter(Parameter != "Median RT") |>
    mutate(Illusion_Difference = round(Illusion_Difference, 5),
           Illusion_DifferenceZ = round(Illusion_DifferenceZ, 6)) |>
    filter(Illusion_Difference %in% coarse_levels(Illusion_Difference))

  gap <- curves |>
    select(Parameter, Type, Illusion_StrengthZ, Illusion_DifferenceZ, Sampler, Predicted, SE) |>
    pivot_wider(names_from = Sampler, values_from = c(Predicted, SE)) |>
    summarise(.by = Parameter,
              Gap = max(abs(Predicted_stanli - Predicted_cmdstanr) / SE_cmdstanr, na.rm = TRUE))
  curves <- curves |>
    left_join(gap, by = "Parameter") |>
    mutate(Panel = sprintf("%s (gap %.2f SD)", Parameter, Gap),
           Illusion_Difference = factor(round(Illusion_Difference, 5)))

  curve_cols <- colorRampPalette(
    c("#FF5722", "#FFC107", "#8BC34A", "#4CAF50")
  )(nlevels(curves$Illusion_Difference))

  ggplot(curves, aes(Illusion_Strength, Predicted, colour = Illusion_Difference,
                     group = interaction(Sampler, Illusion_Difference))) +
    geom_line(aes(linetype = Sampler), linewidth = 0.9) +
    facet_wrap(~Panel, scales = "free_y") +
    scale_colour_manual(values = curve_cols) +
    scale_linetype_manual(values = c("cmdstanr" = "solid", "stanli" = "dashed")) +
    labs(title = label, x = "Illusion Strength", y = NULL, colour = "Task Difficulty") +
    theme_minimal()
}

table_sampler_speed <- function(twin_preds) {
  rows <- purrr::imap_dfr(twin_preds, \(pair, label) {
    time_c <- median(pair$cmdstanr$efficiency$Time_total)
    purrr::imap_dfr(pair, \(p, sampler) {
      e <- p$efficiency
      data.frame(
        Model = label,
        Family = sub(" .*", "", label),
        Sampler = sampler,
        Chains = nrow(e),
        "Step size (median)" = formatC(median(p$convergence$stepsize_chain), format = "f", digits = 4),
        "Warmup (h)" = sprintf("%.0f", median(e$Time_warmup) / 3600),
        "Sampling (h)" = sprintf("%.0f", median(e$Time_sample) / 3600),
        "Total (h)" = sprintf("%.0f", median(e$Time_total) / 3600),
        "Against cmdstanr" = sprintf("%.2fx", median(e$Time_total) / time_c),
        "Bulk ESS / h" = sprintf("%.1f", mean(e$ESS_per_hour)),
        check.names = FALSE
      )
    })
  })
  tinytable::tt(select(rows, -Family)) |>
    tinytable::style_tt(j = 3:9, align = "r") |>
    fill_cells(1, model_fill(rows$Family)) |>
    tinytable::group_tt(j = list("Per chain, median" = 4:7))
}
